Project Overview

About the Project

This chat application is designed to provide functionality similar to Discord, allowing users to:

  • Register and login to the platform
  • Create or join servers
  • Join channels for text chat or voice calls
  • Communicate with other users in real-time
  • Share media and files
  • Manage user roles and permissions
  • Customize user profiles and settings

Technologies Used: HTML, CSS, Bootstrap, JavaScript, PHP, MySQL, GitHub

Recommendation

Consider implementing WebSockets for real-time communication instead of traditional AJAX polling. This will significantly improve the responsiveness of your chat application and reduce server load.

Recommended libraries:

  • Ratchet (PHP WebSocket library)
  • Socket.io (JavaScript client)

Database: Stores all application data including user information, messages, server details, etc. Uses MySQL for relational data storage.

Architecture Recommendations

Consider implementing a layered architecture for better code organization:

  • Presentation Layer: HTML templates, CSS, JavaScript
  • Application Layer: PHP controllers, business logic
  • Data Access Layer: Database queries, models
  • Service Layer: External APIs, WebSockets

This separation of concerns will make your codebase more maintainable and easier to extend.

// User registration function
function registerUser($username, $email, $password) {
    global $conn;
    
    // Validate input
    if (empty($username) || empty($email) || empty($password)) {
        return ['success' => false, 'message' => 'All fields are required'];
    }
    
    // Check if username or email already exists
    $stmt = $conn->prepare("SELECT user_id FROM users WHERE username = ? OR email = ?");
    $stmt->bind_param("ss", $username, $email);
    $stmt->execute();
    $result = $stmt->get_result();
    
    if ($result->num_rows > 0) {
        return ['success' => false, 'message' => 'Username or email already exists'];
    }
    
    // Hash password
    $hashedPassword = password_hash($password, PASSWORD_DEFAULT);
    
    // Insert new user
    $stmt = $conn->prepare("INSERT INTO users (username, email, password, created_at) VALUES (?, ?, ?, NOW())");
    $stmt->bind_param("sss", $username, $email, $hashedPassword);
    
    if ($stmt->execute()) {
        return ['success' => true, 'user_id' => $conn->insert_id];
    } else {
        return ['success' => false, 'message' => 'Registration failed: ' . $conn->error];
    }
}

// User login function
function loginUser($email, $password) {
    global $conn;
    
    // Validate input
    if (empty($email) || empty($password)) {
        return ['success' => false, 'message' => 'Email and password are required'];
    }
    
    // Get user by email
    $stmt = $conn->prepare("SELECT user_id, username, password FROM users WHERE email = ?");
    $stmt->bind_param("s", $email);
    $stmt->execute();
    $result = $stmt->get_result();
    
    if ($result->num_rows === 0) {
        return ['success' => false, 'message' => 'Invalid email or password'];
    }
    
    $user = $result->fetch_assoc();
    
    // Verify password
    if (password_verify($password, $user['password'])) {
        // Update last login time
        $updateStmt = $conn->prepare("UPDATE users SET last_login = NOW(), status = 'online' WHERE user_id = ?");
        $updateStmt->bind_param("i", $user['user_id']);
        $updateStmt->execute();
        
        // Start session
        $_SESSION['user_id'] = $user['user_id'];
        $_SESSION['username'] = $user['username'];
        
        return ['success' => true, 'user_id' => $user['user_id'], 'username' => $user['username']];
    } else {
        return ['success' => false, 'message' => 'Invalid email or password'];
    }
}

// Create a new server
function createServer($name, $description, $owner_id, $icon = null) {
    global $conn;
    
    // Generate a unique invite code
    $invite_code = generateUniqueInviteCode();
    
    // Insert server
    $stmt = $conn->prepare("INSERT INTO servers (name, description, owner_id, icon, invite_code, created_at) 
                           VALUES (?, ?, ?, ?, ?, NOW())");
    $stmt->bind_param("ssiss", $name, $description, $owner_id, $icon, $invite_code);
    
    if ($stmt->execute()) {
        $server_id = $conn->insert_id;
        
        // Add owner as a member with admin role
        addServerMember($server_id, $owner_id, 1); // 1 = admin role
        
        // Create default channels
        createChannel($server_id, 'general', 'text', 'General discussion');
        createChannel($server_id, 'voice-chat', 'voice', 'Voice chat channel');
        
        return ['success' => true, 'server_id' => $server_id, 'invite_code' => $invite_code];
    } else {
        return ['success' => false, 'message' => 'Failed to create server: ' . $conn->error];
    }
}

// Join a server using invite code
function joinServer($user_id, $invite_code) {
    global $conn;
    
    // Find server by invite code
    $stmt = $conn->prepare("SELECT server_id FROM servers WHERE invite_code = ?");
    $stmt->bind_param("s", $invite_code);
    $stmt->execute();
    $result = $stmt->get_result();
    
    if ($result->num_rows === 0) {
        return ['success' => false, 'message' => 'Invalid invite code'];
    }
    
    $server = $result->fetch_assoc();
    $server_id = $server['server_id'];
    
    // Check if user is already a member
    $checkStmt = $conn->prepare("SELECT member_id FROM server_members WHERE server_id = ? AND user_id = ?");
    $checkStmt->bind_param("ii", $server_id, $user_id);
    $checkStmt->execute();
    
    if ($checkStmt->get_result()->num_rows > 0) {
        return ['success' => false, 'message' => 'You are already a member of this server'];
    }
    
    // Add user as member with default role
    return addServerMember($server_id, $user_id, 2); // 2 = default member role
}

// Send a message to a channel
function sendMessage($channel_id, $user_id, $content, $attachment = null) {
    global $conn;
    
    // Check if user has permission to send messages in this channel
    if (!canUserAccessChannel($user_id, $channel_id)) {
        return ['success' => false, 'message' => 'You do not have permission to send messages in this channel'];
    }
    
    // Insert message
    $stmt = $conn->prepare("INSERT INTO messages (channel_id, user_id, content, attachment, created_at) 
                           VALUES (?, ?, ?, ?, NOW())");
    $stmt->bind_param("iiss", $channel_id, $user_id, $content, $attachment);
    
    if ($stmt->execute()) {
        $message_id = $conn->insert_id;
        
        // Get message details for real-time update
        $messageData = getMessageById($message_id);
        
        // Notify channel subscribers via WebSocket
        notifyChannelSubscribers($channel_id, 'new_message', $messageData);
        
        return ['success' => true, 'message_id' => $message_id, 'data' => $messageData];
    } else {
        return ['success' => false, 'message' => 'Failed to send message: ' . $conn->error];
    }
}

// Edit an existing message
function editMessage($message_id, $user_id, $new_content) {
    global $conn;
    
    // Check if user is the message author
    $stmt = $conn->prepare("SELECT channel_id FROM messages WHERE message_id = ? AND user_id = ?");
    $stmt->bind_param("ii", $message_id, $user_id);
    $stmt->execute();
    $result = $stmt->get_result();
    
    if ($result->num_rows === 0) {
        return ['success' => false, 'message' => 'You can only edit your own messages'];
    }
    
    $message = $result->fetch_assoc();
    $channel_id = $message['channel_id'];
    
    // Update message
    $updateStmt = $conn->prepare("UPDATE messages SET content = ?, is_edited = 1, updated_at = NOW() WHERE message_id = ?");
    $updateStmt->bind_param("si", $new_content, $message_id);
    
    if ($updateStmt->execute()) {
        // Get updated message for real-time update
        $messageData = getMessageById($message_id);
        
        // Notify channel subscribers via WebSocket
        notifyChannelSubscribers($channel_id, 'edit_message', $messageData);
        
        return ['success' => true, 'data' => $messageData];
    } else {
        return ['success' => false, 'message' => 'Failed to edit message: ' . $conn->error];
    }
}
